  globalThis[MARK] = true;
  if (document.readyState === "loading") {
    document.addEventListener("DOMContentLoaded", init, { once: true });
  } else {
    init();
  }

  const parseJson = (value, label = "JSON value") => {
    try {
      return JSON.parse(value);
    } catch {
      throw new Error(`Invalid ${label}: ${value}`);
    }
  };
  const api = async (path, options = {}) => {
    const response = await fetch(path, {
      ...options,
      headers: {
        ...(options.body ? { "Content-Type": "application/json" } : {}),
        ...options.headers,
      },
    });
    const payload = await response.json().catch(() => ({ detail: response.statusText }));
    if (!response.ok) throw new Error(payload.detail || `Request failed (${response.status})`);
    return payload;
  };

  function freshTemplate(inputs = ["input"]) {
    return { input_names: [...inputs], nodes: [], outputs: { output: inputs[0] } };
  }

  function freshDocument() {
    return {
      format: "research-assistant/torch-architecture",
      version: 2,
      name: "model",
      description: "",
      graph: {
        variables: {},
        variable_specs: {},
        ...freshTemplate(),
        subgraphs: {},
      },
    };
  }

  function normalizeDocument(value) {
    if (!value || typeof value !== "object") throw new Error("Architecture must be an object");
    if (value.format && value.format !== "research-assistant/torch-architecture") {
      throw new Error("Unsupported architecture format");
    }
    const graph = value.graph || value;
    const outputs = Array.isArray(graph.outputs)
      ? Object.fromEntries(
          graph.outputs.map((source, index) => [graph.outputs.length === 1 ? "output" : `output_${index}`, source]),
        )
      : graph.outputs || { output: graph.input_names?.[0] || "input" };
    const subgraphs = Object.fromEntries(
      Object.entries(graph.subgraphs || {}).map(([name, template]) => [
        name,
        {
          input_names: template.input_names || ["input"],
          nodes: (template.nodes || []).map(normalizeNode),
          outputs: Array.isArray(template.outputs)
            ? Object.fromEntries(
                template.outputs.map((source, index) => [
                  template.outputs.length === 1 ? "output" : `output_${index}`,
                  source,
                ]),
              )
            : template.outputs || { output: template.input_names?.[0] || "input" },
        },
      ]),
    );
    return {
      format: "research-assistant/torch-architecture",
      version: 2,
      name: value.name || "model",
      description: value.description || "",
      graph: {
        variables: graph.variables || {},
        variable_specs: graph.variable_specs || {},
        input_names: graph.input_names || ["input"],
        nodes: (graph.nodes || []).map(normalizeNode),
        outputs,
        subgraphs,
      },
    };
  }

  function normalizeNode(node) {
    const kind = node.kind || "module";
    const outputPorts = node.output_ports || ["output"];
    return {
      id: node.id,
      kind,
      type: node.type ?? null,
      target: node.target ?? null,
      template: node.template ?? null,
      inputs: deepCopy(node.inputs || ["input"]),
      params: deepCopy(node.params || {}),
      output_ports: [...outputPorts],
      call_style: node.call_style || "auto",
      count: node.count ?? null,
      weights: node.weights || "independent",
      index_name: node.index_name || "index",
      carry: deepCopy(node.carry || {}),
      selector: node.selector ?? null,
      branches: deepCopy(node.branches || {}),
      default_branch: node.default_branch ?? null,
      label: node.label ?? null,
      position: node.position || { x: 260, y: 70 },
    };
  }
