   if (schema.type === "boolean") {
      const select = selectOptions(["true", "false"], String(Boolean(value)));
      select.onchange = () => onChange(select.value === "true");
      return select;
    }
    if (schema.enum) {
      const select = element("select");
      for (const choice of schema.enum) {
        const option = element("option", null, String(choice));
        option.value = JSON.stringify(choice);
        select.append(option);
      }
      select.value = JSON.stringify(value);
      select.onchange = () => onChange(JSON.parse(select.value));
      return select;
    }
    const complex = schema.type === "array" || schema.type === "object" || typeof value === "object";
    const input = element(complex ? "textarea" : "input");
    if (complex) input.className = "ra-json-editor";
    input.value = complex ? JSON.stringify(value ?? schema.default ?? null) : String(value ?? schema.default ?? "");
    input.onchange = () => onChange(parseLiteral(input.value, schema));
    return input;
  }

  function bindingEditor(value, schema, onChange) {
    const host = element("div", "ra-binding");
    const mode = selectOptions(["literal", "variable", "expression"]);
    let selectedMode = "literal";
    if (value && typeof value === "object" && Object.keys(value).length === 1) {
      if (value.$var) selectedMode = "variable";
      if (value.$expr) selectedMode = "expression";
    }
    mode.value = selectedMode;
    const renderValue = () => {
      host.firstElementChild?.remove();
      let editor;
      if (mode.value === "variable") {
        editor = selectOptions(Object.keys(graph().variables).sort(), value?.$var || Object.keys(graph().variables)[0] || "");
        editor.onchange = () => onChange({ $var: editor.value });
      } else if (mode.value === "expression") {
        editor = element("input");
        editor.placeholder = "e.g. layer_steps[layer_index]";
        editor.value = value?.$expr || "";
        editor.onchange = () => onChange({ $expr: editor.value.trim() });
      } else {
        const literal = value && typeof value === "object" && (value.$var || value.$expr) ? schema.default : value;
        editor = literalEditor(literal, schema, onChange);
      }
      host.insertBefore(editor, mode);
    };
    mode.onchange = () => {
      if (mode.value === "variable") {
        const first = Object.keys(graph().variables)[0];
        if (!first) {
          mode.value = "literal";
          throw new Error("Add an architecture variable first");
        }
        value = { $var: first };
        onChange(value);
      } else if (mode.value === "expression") {
        value = { $expr: "" };
        onChange(value);
      } else {
        value = deepCopy(schema.default ?? "");
        onChange(value);
      }
      renderValue();
    };
    host.append(element("span"), mode);
    renderValue();
    return host;
  }

  function renderInputs(node) {
    ui.inspector.append(element("div", "ra-inspector-title", "Named inputs"));
    if (Array.isArray(node.inputs)) {
      node.inputs = Object.fromEntries(node.inputs.map((source, index) => [index === 0 ? "input" : `input_${index + 1}`, source]));
    }
    for (const [port, source] of Object.entries(node.inputs)) {
      const row = element("div", "ra-input-row");
      const portInput = element("input");
      portInput.value = port;
      const sourceSelect = selectOptions(availableSources(node.id), source);
      const remove = element("button", "icon-button", "×");
      portInput.onchange = () => {
        const next = portInput.value.trim();
        if (!/^[A-Za-z][A-Za-z0-9_]*$/.test(next)) throw new Error("Invalid input port");
        const entries = Object.entries(node.inputs).map(([name, value]) => [name === port ? next : name, value]);
        node.inputs = Object.fromEntries(entries);
        renderInspector();
        drawEdges();
      };
      sourceSelect.onchange = () => {
        node.inputs[port] = sourceSelect.value;
        drawEdges();
      };
      remove.onclick = () => {
        delete node.inputs[port];
        renderInspector();
        drawEdges();
      };
      row.append(portInput, sourceSelect, remove);
      ui.inspector.append(row);
    }
    const add = element("button", "button compact ghost", "Add input port");
    add.onclick = () => {
      let index = Object.keys(node.inputs).length + 1;
      let name = `input_${index}`;
      while (Object.hasOwn(node.inputs, name)) name = `input_${++index}`;
      node.inputs[name] = availableSources(node.id)[0] || currentTemplate().input_names[0];
      renderInspector();
      drawEdges();
    };
    ui.inspector.append(add);
  }

  function renderOutputs(node) {
    ui.inspector.append(element("div", "ra-inspector-title", "Output ports"));
    const input = element("input");
    input.value = node.output_ports.join(", ");
    input.onchange = () => {
      const ports = input.value.split(",").map((item) => item.trim()).filter(Boolean);
      if (!ports.length || ports.some((item) => !/^[A-Za-z][A-Za-z0-9_]*$/.test(item))) {
        throw new Error("Output ports must be comma-separated identifiers");
      }
      node.output_ports = [...new Set(ports)];
      renderGraph();
    };
    ui.inspector.append(field("Ports", input));
  }

  function renderModuleParameters(node) {
    const spec = moduleSpec(node.type);
    ui.inspector.append(element("div", "ra-inspector-title", "Module parameters"));
    for (const [name, schema] of Object.entries(spec?.schema?.properties || {})) {
      const binding = bindingEditor(node.params[name], schema, (value) => (node.params[name] = value));
      ui.inspector.append(field(name, binding));
    }
    const callStyle = selectOptions(["auto", "positional", "keyword"], node.call_style);
    callStyle.onchange = () => (node.call_style = callStyle.value);
    ui.inspector.append(field("Call style", callStyle));
  }

  }

  function templateSelect(selected, allowEmpty = true) {
    const values = allowEmpty ? [""] : [];
    values.push(...templateNames());
    return selectOptions(values, selected || "");
  }
