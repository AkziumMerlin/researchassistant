uped.has(category)) grouped.set(category, []);
      grouped.get(category).push(item);
    }
    for (const [category, specs] of [...grouped].sort(([a], [b]) => a.localeCompare(b))) {
      ui.paletteList.append(element("div", "ra-palette-group", category));
      for (const spec of specs) {
        const button = element("button", "ra-palette-button");
        button.append(
          element("strong", null, spec.name.split("/").at(-1)),
          element("small", null, spec.description),
        );
        button.onclick = () => addModuleNode(spec);
        ui.paletteList.append(button);
      }
    }
  }

  function nextNodeId(base) {
    const clean = base.replace(/\W/g, "_").replace(/^_+/, "").toLowerCase() || "node";
    const ids = new Set(currentTemplate().nodes.map((node) => node.id));
    let candidate = clean;
    let index = 2;
    while (ids.has(candidate)) candidate = `${clean}_${index++}`;
    return candidate;
  }

  function availableSources(exclude = null) {
    const template = currentTemplate();
    const sources = [...template.input_names];
    for (const node of template.nodes) {
      if (node.id === exclude) continue;
      for (const port of node.output_ports || ["output"]) sources.push(`${node.id}.${port}`);
      if ((node.output_ports || ["output"]).length === 1) sources.push(node.id);
    }
    return [...new Set(sources)];
  }

  function defaultSource() {
    const template = currentTemplate();
    const last = template.nodes.at(-1);
    if (last) return last.output_ports.length === 1 ? last.id : `${last.id}.${last.output_ports[0]}`;
    return template.input_names[0];
  }

  function addModuleNode(spec) {
    const count = spec.metadata?.inputs ?? spec.metadata?.min_inputs ?? 1;
    const inputs = {};
    for (let index = 0; index < count; index++) inputs[index === 0 ? "input" : `input_${index + 1}`] = defaultSource();
    const params = Object.fromEntries(
      Object.entries(spec.schema?.properties || {})
        .filter(([, schema]) => schema.default !== undefined)
        .map(([name, schema]) => [name, deepCopy(schema.default)]),
    );
    const node = normalizeNode({
      id: nextNodeId(spec.name.split("/").at(-1)),
      kind: "module",
      type: spec.name,
      inputs,
      params,
      output_ports: ["output"],
      call_style: "positional",
      position: nextPosition(),
    });
    appendNode(node);
  }

  function addControlNode(kind) {
    const firstTemplate = templateNames()[0] || null;
    const common = {
      id: nextNodeId(kind),
      kind,
      inputs: { input: defaultSource() },
      output_ports: ["output"],
      position: nextPosition(),
    };
    if (kind === "python") Object.assign(common, { target: "torch.nn:Identity", params: {}, call_style: "positional" });
    if (kind === "composite") Object.assign(common, { template: firstTemplate });
    if (kind === "repeat") {
      Object.assign(common, {
        template: firstTemplate,
        count: 1,
        weights: "independent",
        index_name: "index",
        carry: {},
      });
    }
    if (kind === "switch") Object.assign(common, { selector: true, branches: {}, default_branch: firstTemplate });
    appendNode(normalizeNode(common));
  }

  function nextPosition() {
    const count = currentTemplate().nodes.length;
    return { x: 270 + (count % 6) * 220, y: 70 + Math.floor(count / 6) * 135 };
  }

  function appendNode(node) {
    currentTemplate().nodes.push(node);
    const outputs = currentTemplate().outputs;
    if (Object.keys(outputs).length === 1 && Object.values(outputs)[0] === currentTemplate().input_names[0]) {
      currentTemplate().outputs = { output: node.id };
    }
    state.selectedNode = node.id;
    renderGraph();
  }

  function graphUsesNodeAsOutput(node) {
    return Object.values(currentTemplate().outputs).some((source) => source === node.id || source.startsWith(`${node.id}.`));
  }

  function renderGraph() {
    renderTemplateNavigation();
    const template = currentTemplate();
    ui.nodes.replaceChildren();
    for (const [index, name] of template.input_names.entries()) {
      drawNode({ id: name, kind: "input", output_ports: ["output"], position: { x: 25, y: 60 + index * 100 } }, true);
    }
    for (const node of template.nodes) drawNode(node, false);
    drawEdges();
    renderInspector();
    ui.status.textContent = `${template.nodes.length} node(s) · ${templateNames().length} subgraph(s) · not validated`;
    ui.status.classList.remove("ra-models-valid");
  }

  function drawNode(node, isInput) {
    const card = element(
      "div",
      `ra-node-v2${isInput ? " input" : ""}${state.selectedNode === node.id ? " selected" : ""}${!isInput && graphUsesNodeAsOutput(node) ? " output" : ""}`,
    );
    card.style.left = `${node.position.x}px`;
    card.style.top = `${node.position.y}px`;
    const title = element("h4", null, node.label || node.id);
    const subtitle = element("small", null, isInput ? "model input" : nodeKindLabel(node));
    const ports = element("div", "ra-node-ports", `out: ${(node.output_ports || ["output"]).join(", ")}`);
    card.append(title, subtitle, ports);
    if (!isInput) {
      card.onclick = () => {
        state.selectedNode = node.id;
        renderGraph();
      };
      enableDrag(title, node, card);
    }
    ui.nodes.append(card);
  }

  function nodeKindLabel(node) {
    if (node.kind === "module") return node.type;
    if (node.kind === "python") return node.target || "Python module";
    if (node.kind === "composite") return `Composite · ${node.template || "unset"}`;
    if (node.kind === "repeat") return `Repeat ${bindingText(node.count)} · ${node.template || "unset"}`;
    if (node.kind === "switch") return `Switch · ${bindingText(node.selector)}`;
    return node.kind;
  }

  function bindingText(value) {
    if (value && typeof value === "object" && Object.keys(value).length === 1) {
      if (value.$var) return `$${value.$var}`;
      if (value.$expr) return value.$expr;
    }
    return JSON.stringify(value);
  }

  function enableDrag(handle, node, card) {
    handle.onpointerdown = (event) => {
      handle.setPointerCapture(event.pointerId);
      const startX = event.clientX;
      const startY = event.clientY;
      const originX = node.position.x;
      const originY = node.position.y;
      handle.onpointermove = (move) => {
        node.position.x = Math.max(5, originX + move.clientX - startX);
        node.position.y = Math.max(5, originY + move.clientY - startY);
        card.style.left = `${node.position.x}px`;
        card.style.top = `${node.position.y}px`;
        drawEdges();
      };
      handle.onpointerup = () => (handle.onpointermove = null);
    };
  }

  function drawEdges() {
    ui.edges.replaceChildren();
    const positions = new Map(
      currentTemplate().input_names.map((name, index) => [name, { x: 25, y: 60 + index * 100 }]),
    );
    for (const node of currentTemplate().nodes) positions.set(node.id, node.position);
    for (const node of currentTemplate().nodes) {
      const sources = Array.isArray(node.inputs) ? node.inputs : Object.values(node.inputs);
      for (const source of sources) {
        const owner = source.split(".", 1)[0];
        const from = positions.get(owner);
        if (!from) continue;
        const path = document.createElementNS("http://www.w3.org/2000/svg", "path");
        const x1 = from.x + 195;
        const y1 = from.y + 35;
        const x2 = node.position.x;
        const y2 = node.position.y + 35;
        const bend = Math.max(45, Math.abs(x2 - x1) * 0.4);
        path.setAttribute("d", `M${x1} ${y1} C${x1 + bend} ${y1},${x2 - bend} ${y2},${x2} ${y2}`);
        path.setAttribute("class", "ra-edge");
        ui.edges.append(path);
      }
    }
  }

  function field(label, input) {

  function selectOptions(values, selected = null) {
    const select = element("select");
    for (const value of values) {
      const option = element("option", null, value);
      option.value = value;
      select.append(option);
    }
    if (selected !== null) select.value = selected;
    return select;
  }

  function parseLiteral(raw, schema = {}) {
    if (schema.type === "integer") {
      const value = Number(raw);
      if (!Number.isInteger(value)) throw new Error("Expected integer");
      return value;
    }
    if (schema.type === "number") {
      const value = Number(raw);
      if (!Number.isFinite(value)) throw new Error("Expected number");
      return value;
    }
    if (schema.type === "boolean") return raw === "true";
    if (schema.type === "array" || schema.type === "object") return parseJson(raw);
    if (schema.enum) return parseJson(raw);
    if (schema.anyOf) {
      try {
        return parseJson(raw);
      } catch {
        return raw;
      }
    }
    return raw;
  }
