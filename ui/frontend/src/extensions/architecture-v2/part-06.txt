tring(choice));
        option.value = JSON.stringify(choice);
        select.append(option);
      }
      select.value = JSON.stringify(value);
      select.onchange = () => onChange(JSON.parse(select.value));
      return select;
    }
    const input = element("input");
    if (spec.type === "int" || spec.type === "float") {
      input.type = "number";
      input.step = spec.type === "int" ? "1" : "any";
      input.value = String(value);
      input.onchange = () => onChange(Number(input.value));
    } else if (spec.type === "string") {
      input.value = value || "";
      input.onchange = () => onChange(input.value);
    } else {
      input.value = JSON.stringify(value);
      input.onchange = () => onChange(parseJson(input.value));
    }
    return input;
  }

  function renderConfigArchitecture(item) {
    const select = q(".component-type", item);
    const host = q(".schema-fields", item);
    if (!select || !host || select.value !== "torch/parameterized-graph") return;
    if (q(".ra-config-architecture", host)) return;
    host.dataset.editor = "torch-graph";
    const card = element("div", "ra-config-architecture");
    const picker = element("select");
    const placeholder = element("option", null, "Select saved architecture…");
    placeholder.value = "";
    picker.append(placeholder);
    for (const file of state.files) {
      const option = element("option", null, file.path);
      option.value = file.path;
      picker.append(option);
    }
    const open = element("button", "button compact ghost", "Open Models");
    const variables = element("div");
    const note = element("div", "architecture-empty", "Select an architecture; variables remain matrix/override targets.");
    open.onclick = () => openModels().catch(showError);
    picker.onchange = async () => {
      if (!picker.value) return;
      try {
        const file = await api(`/api/files?path=${encodeURIComponent(picker.value)}`);
        const documentValue = normalizeDocument(JSON.parse(file.content));
        const graphValue = documentValue.graph;
        host.dataset.graph = JSON.stringify(graphValue);
        const rerender = () => {
          variables.replaceChildren();
          for (const name of Object.keys(graphValue.variables).sort()) {
            const spec = graphValue.variable_specs?.[name] || {};
            const row = element("div", "ra-config-variable");
            const title = element("strong", null, name);
            const input = configVariableInput(name, graphValue, (value) => {
              graphValue.variables[name] = value;
              host.dataset.graph = JSON.stringify(graphValue);
              rerender();
            });
            const path = element("code", null, `components.model.params.variables.${name}`);
            if (!evaluateEnabled(spec.enabled_if, graphValue.variables)) {
              row.classList.add("disabled");
              input.disabled = true;
            }
            row.append(title, input, path);
            variables.append(row);
          }
        };
        rerender();
        note.textContent = `${graphValue.nodes.length} root nodes · ${Object.keys(graphValue.subgraphs || {}).length} subgraphs · ${Object.keys(graphValue.variables).length} variables`;
      } catch (error) {
        note.textContent = error.message;
      }
    };
    card.append(picker, open, variables, note);
    host.replaceChildren(card);
  }

  function enhanceConfigCreator() {
    const root = q("#creator-components");
    if (!root) return;
    for (const item of root.querySelectorAll(".component-item")) {
      const select = q(".component-type", item);
      if (!select) continue;
      const option = [...select.options].find((candidate) => candidate.value === "torch/parameterized-graph");
      if (option) option.textContent = "torch/parameterized-graph (advanced saved architecture)";
      if (!select.dataset.raArchitectureV2) {
        select.dataset.raArchitectureV2 = "1";
        select.addEventListener("change", () => queueMicrotask(() => renderConfigArchitecture(item)));
      }
      renderConfigArchitecture(item);
    }
  }
