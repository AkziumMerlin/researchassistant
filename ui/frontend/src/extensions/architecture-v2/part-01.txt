  function addSubgraph() {
    const requested = window.prompt("Subgraph name", `block_${templateNames().length + 1}`);
    if (!requested) return;
    const name = requested.trim();
    if (!/^[A-Za-z][A-Za-z0-9_]*$/.test(name)) throw new Error("Invalid subgraph name");
    if (graph().subgraphs[name]) throw new Error(`Subgraph ${name} already exists`);
    graph().subgraphs[name] = freshTemplate();
    selectTemplate(name);
    renderPalette();
  }

  function deleteSubgraph() {
    if (state.activeTemplate === ROOT) return;
    const name = state.activeTemplate;
    const used = allNodes().filter(
      (node) =>
        node.template === name ||
        Object.values(node.branches || {}).includes(name) ||
        node.default_branch === name,
    );
    if (used.length) throw new Error(`Subgraph ${name} is used by: ${used.map((node) => node.id).join(", ")}`);
    if (!window.confirm(`Delete subgraph ${name}?`)) return;
    delete graph().subgraphs[name];
    selectTemplate(ROOT);
  }

  function allNodes() {
    return [graph().nodes, ...Object.values(graph().subgraphs).map((item) => item.nodes)].flat();
  }

  function defaultValue(type) {
    if (type === "bool") return false;
    if (type === "int") return 1;
    if (type === "float") return 0.0;
    if (type === "string") return "";
    if (type === "enum") return "option";
    if (type === "shape") return [1];
    return null;
  }
