  function decorateDialog(dialog) {
    if (!dialog || dialog.dataset.raThemeReady === "1") return;
    dialog.dataset.raThemeReady = "1";
    const header = dialog.querySelector(".raH,.raPipeH,.raRH");
    if (header) {
      const title = header.querySelector("b");
      if (title) {
        const wrapper = document.createElement("div");
        wrapper.className = "ra-dialog-title";
        const eyebrow = document.createElement("span");
        eyebrow.className = "eyebrow";
        const heading = document.createElement("h2");
        const metadata = {
          "ra-jobs": ["EXECUTION", "Jobs and live metrics"],
          "ra-charts": ["ANALYTICS", "Advanced charts"],
          "ra-pipeline": ["OPERATIONS", "Research pipeline"],
          "ra-research": ["RESEARCH", "End-to-end research"],
        }[dialog.id] || ["WORKBENCH", title.textContent];
        eyebrow.textContent = metadata[0];
        heading.textContent = metadata[1];
        wrapper.append(eyebrow, heading);
        title.replaceWith(wrapper);
      }
    }
    dialog.querySelectorAll(".raB,.raPipeB,.raRB").forEach((button) => {
      const label = button.textContent.trim();
      if (PRIMARY_ACTIONS.has(label)) button.classList.add("ra-theme-primary");
    });
  }

  const REGISTRY_STATE_KEY = "research-assistant.registry-open.v1";

  function registryIsOpen() {
    return document.querySelector(".workbench")?.classList.contains("ra-registry-open") || false;
  }

  function updateRegistryToggle() {
    const toggle = document.querySelector("#ra-registry-toggle");
    if (!toggle) return;
    const open = registryIsOpen();
    toggle.textContent = open ? "Hide Registry" : "Registry";
    toggle.setAttribute("aria-pressed", String(open));
  }

  function setRegistryOpen(open, persist = true) {
    const workbench = document.querySelector(".workbench");
    const panel = document.querySelector(".registry-panel");
    if (!workbench || !panel) return;
    const changed = registryIsOpen() !== open || panel.hidden === open;
    workbench.classList.toggle("ra-registry-open", open);
    panel.hidden = !open;
    panel.setAttribute("aria-hidden", String(!open));
    if (persist) {
      try { localStorage.setItem(REGISTRY_STATE_KEY, open ? "1" : "0"); } catch {}
    }
    updateRegistryToggle();
    if (changed) requestAnimationFrame(() => window.dispatchEvent(new Event("resize")));
  }

  function prepareRegistryPanel() {
    const panel = document.querySelector(".registry-panel");
    const heading = panel?.querySelector(".panel-heading");
    if (!panel || !heading) return;
    let controls = heading.querySelector(".ra-registry-controls");
    if (!controls) {
      controls = document.createElement("span");
      controls.className = "ra-registry-controls";
      const count = heading.querySelector("#component-count");
      if (count) controls.append(count);
      const close = document.createElement("button");
      close.type = "button";
      close.className = "icon-button";
      close.setAttribute("aria-label", "Hide Registry");
      close.textContent = "×";
      close.addEventListener("click", () => setRegistryOpen(false));
      controls.append(close);
      heading.append(controls);
    }
    let open = false;
    try { open = localStorage.getItem(REGISTRY_STATE_KEY) === "1"; } catch {}
    setRegistryOpen(open, false);
  }

  function normalizeTopbar() {
    const actions = document.querySelector(".topbar-actions");
    if (!actions) return;
    const rename = new Map([
      ["Jobs+", "Jobs"],
      ["Charts+", "Charts"],
      ["Pipeline+", "Pipeline"],
      ["Research+", "Research"],
    ]);
    [...actions.querySelectorAll("button")].forEach((button) => {
      const label = button.textContent.trim();
      if (rename.has(label)) button.textContent = rename.get(label);
    });

    let nav = actions.querySelector(".ra-section-nav");
    if (!nav) {
      nav = document.createElement("nav");
      nav.className = "ra-section-nav";
      nav.setAttribute("aria-label", "Research workspaces");
    }
    let more = actions.querySelector(".ra-more-menu");
    if (!more) {
      more = document.createElement("details");
      more.className = "ra-more-menu";
      const summary = document.createElement("summary");
      summary.className = "button ghost";
      summary.textContent = "More";
      const popover = document.createElement("div");
      popover.className = "ra-more-popover";
      more.append(summary, popover);
    }
    const popover = more.querySelector(".ra-more-popover");
    if (popover.dataset.raCloseBound !== "1") {
      popover.dataset.raCloseBound = "1";
      popover.addEventListener("click", (event) => {
        if (event.target.closest("button")) more.removeAttribute("open");
      });
    }
    let registryToggle = actions.querySelector("#ra-registry-toggle");
    if (!registryToggle) {
      registryToggle = document.createElement("button");
      registryToggle.id = "ra-registry-toggle";
      registryToggle.type = "button";
      registryToggle.className = "button ghost";
      registryToggle.textContent = "Registry";
      registryToggle.addEventListener("click", () => setRegistryOpen(!registryIsOpen()));
      actions.append(registryToggle);
    }
    updateRegistryToggle();

    const primaryLabels = ["Models", "Launch", "Jobs", "Pipeline", "Research", "Reports"];
    const moreLabels = ["Registry", "Charts", "Checkpoints", "Project", "New file"];
    const buttons = [...actions.querySelectorAll(":scope > button")];
    const byLabel = new Map(buttons.map((button) => [button.textContent.trim(), button]));

    for (const label of primaryLabels) {
      const button = byLabel.get(label);
      if (button) nav.append(button);
    }
    for (const label of moreLabels) {
      const button = byLabel.get(label);
      if (button) {
        popover.append(button);
        button.addEventListener("click", () => more.removeAttribute("open"), { once: false });
      }
    }

    const newConfig = byLabel.get("New config");
    const inspect = byLabel.get("Inspect");
    const save = byLabel.get("Save");
    const handled = new Set([...primaryLabels, ...moreLabels, "New config", "Inspect", "Save"]);
    for (const button of buttons) {
      if (!handled.has(button.textContent.trim())) popover.append(button);
    }
    actions.replaceChildren();
    if (newConfig) actions.append(newConfig);
    actions.append(nav, more);
    if (inspect) actions.append(inspect);
    if (save) actions.append(save);
  }

  function apply() {
    observer.disconnect();
    document.querySelectorAll("dialog.raX,dialog.raPipe,dialog.raResearch,dialog.ra-models-v2")
      .forEach(decorateDialog);
    prepareRegistryPanel();
    normalizeTopbar();
    observer.observe(document.documentElement, { childList: true, subtree: true });
  }

  let pending = false;
  const observer = new MutationObserver((mutations) => {
    const actions = document.querySelector(".topbar-actions");
    const relevant = mutations.some((mutation) => {
      if (mutation.target === actions) return true;
      return [...mutation.addedNodes].some(
        (node) => node.nodeType === Node.ELEMENT_NODE
          && (node.matches?.("dialog") || node.querySelector?.("dialog")),
      );
    });
    if (!relevant || pending) return;
    pending = true;
    requestAnimationFrame(() => {
      pending = false;
      apply();
    });
  });
  apply();
}

function scheduleUnifiedWorkbenchTheme() {
  if (document.readyState === "loading") {
    document.addEventListener("DOMContentLoaded", () => setTimeout(installUnifiedWorkbenchTheme, 0), { once: true });
  } else {
    setTimeout(installUnifiedWorkbenchTheme, 0);
  }
}

scheduleUnifiedWorkbenchTheme();
