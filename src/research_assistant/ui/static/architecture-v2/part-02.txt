bles[newName] = graph().variables[oldName];
    delete graph().variables[oldName];
    graph().variable_specs[newName] = graph().variable_specs[oldName] || { type: "json" };
    delete graph().variable_specs[oldName];
    const escaped = oldName.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
    const word = new RegExp(`\\b${escaped}\\b`, "g");
    const walk = (value) => {
      if (Array.isArray(value)) return value.map(walk);
      if (value && typeof value === "object") {
        if (Object.keys(value).length === 1 && value.$var === oldName) return { $var: newName };
        if (Object.keys(value).length === 1 && typeof value.$expr === "string") {
          return { $expr: value.$expr.replace(word, newName) };
        }
        return Object.fromEntries(Object.entries(value).map(([key, nested]) => [key, walk(nested)]));
      }
      return value;
    };
    for (const node of allNodes()) {
      node.params = walk(node.params);
      node.count = walk(node.count);
      node.selector = walk(node.selector);
    }
    for (const spec of Object.values(graph().variable_specs)) {
      if (spec.enabled_if) spec.enabled_if = spec.enabled_if.replace(word, newName);
    }
  }

  function variableInput(name, spec) {
    const value = graph().variables[name];
    if (spec.type === "bool") {
      const input = element("input");
      input.type = "checkbox";
      input.checked = Boolean(value);
      input.onchange = () => (graph().variables[name] = input.checked);
      return input;
    }
    if (spec.type === "enum") {
      const select = element("select");
      for (const choice of spec.choices || []) {
        const option = element("option", null, String(choice));
        option.value = JSON.stringify(choice);
        select.append(option);
      }
      select.value = JSON.stringify(value);
      select.onchange = () => (graph().variables[name] = JSON.parse(select.value));
      return select;
    }
    const input = element("input");
    if (spec.type === "int" || spec.type === "float") {
      input.type = "number";
      input.step = spec.type === "int" ? "1" : "any";
      if (spec.min !== undefined) input.min = String(spec.min);
      if (spec.max !== undefined) input.max = String(spec.max);
      input.value = String(value);
      input.onchange = () => {
        const parsed = Number(input.value);
        if (!Number.isFinite(parsed) || (spec.type === "int" && !Number.isInteger(parsed))) {
          throw new Error(`Invalid ${spec.type} value for ${name}`);
        }
        graph().variables[name] = parsed;
      };
    } else if (spec.type === "string") {
      input.value = value ?? "";
      input.onchange = () => (graph().variables[name] = input.value);
    } else {
      input.value = format(value);
      input.onchange = () => (graph().variables[name] = parseJson(input.value));
    }
    return input;
  }

  function renderVariables() {
    ui.variablePanel.replaceChildren();
    const heading = element("div", "ra-variable-head");
    heading.append(
      element("strong", null, "Architecture variables"),
      element("span", null, "bool · enum · numeric · shape · JSON; matrix overrides target variables.<name>"),
    );
    ui.variablePanel.append(heading);
    for (const name of Object.keys(graph().variables).sort()) {
      const spec = graph().variable_specs[name] || { type: inferVariableType(graph().variables[name]) };
      graph().variable_specs[name] = spec;
      const row = element("div", "ra-variable-row");
      const nameInput = element("input");
      nameInput.value = name;
      nameInput.onchange = () => {
        renameVariable(name, nameInput.value.trim());
        renderVariables();
        renderInspector();
      };
      const typeSelect = element("select");
      for (const type of ["int", "float", "bool", "string", "enum", "shape", "json"]) {
        const option = element("option", null, type);
        option.value = type;
        typeSelect.append(option);
      }
      typeSelect.value = spec.type || "json";
      typeSelect.onchange = () => {
        spec.type = typeSelect.value;
        if (spec.type === "enum") {
          spec.choices = spec.choices?.length ? spec.choices : ["option"];
          graph().variables[name] = spec.choices[0];
        } else {
          delete spec.choices;
          graph().variables[name] = defaultValue(spec.type);
        }
        renderVariables();
        renderInspector();
      };
      const valueHost = element("div");
      valueHost.append(variableInput(name, spec));
      const meta = element("div", "ra-variable-meta");
      const description = element("input");
      description.placeholder = "description";
      description.value = spec.description || "";
      description.onchange = () => (spec.description = description.value);
      description.className = "wide";
      meta.append(description);
      if (spec.type === "enum") {
        const choices = element("input");
        choices.placeholder = 'choices JSON, e.g. ["a","b"]';
        choices.value = JSON.stringify(spec.choices || []);
        choices.className = "wide";
        choices.onchange = () => {
          const parsed = parseJson(choices.value, "enum choices");
          if (!Array.isArray(parsed) || !parsed.length) throw new Error("Enum choices must be non-empty");
          spec.choices = parsed;
          if (!parsed.some((item) => JSON.stringify(item) === JSON.stringify(graph().variables[name]))) {
            graph().variables[name] = parsed[0];
          }
          renderVariables();
        };
        meta.append(choices);
      }
      if (spec.type === "int" || spec.type === "float") {
        const minimum = element("input");
        minimum.type = "number";
        minimum.placeholder = "min";
        minimum.value = spec.min ?? "";
        minimum.onchange = () => {
          if (minimum.value === "") delete spec.min;
          else spec.min = Number(minimum.value);
        };
        const maximum = element("input");
        maximum.type = "number";
        maximum.placeholder = "max";
        maximum.value = spec.max ?? "";
        maximum.onchange = () => {
          if (maximum.value === "") delete spec.max;
          else spec.max = Number(maximum.value);
        };
        meta.append(minimum, maximum);
      }
      const enabled = element("input");
      enabled.placeholder = 'enabled_if, e.g. backend == "kuramoto"';
      enabled.value = spec.enabled_if || "";
      enabled.className = "wide";
      enabled.onchange = () => {
        if (enabled.value.trim()) spec.enabled_if = enabled.value.trim();
        else delete spec.enabled_if;
      };
      meta.append(enabled);
      const path = element("code", "ra-variable-path", `components.model.params.variables.${name}`);
      const remove = element("button", "icon-button", "×");
      remove.type = "button";
      remove.onclick = () => {
        delete graph().variables[name];
        delete graph().variable_specs[name];
        renderVariables();
        renderInspector();
      };
      row.append(nameInput, typeSelect, valueHost, meta, path, remove);
      ui.variablePanel.append(row);
    }
    if (!Object.keys(graph().variables).length) {
      ui.variablePanel.append(element("div", "architecture-empty", "No variables. Add typed flags, categories or dimensions."));
    }
  }

  function inferVariableType(value) {
    if (typeof value === "boolean") return "bool";
    if (Number.isInteger(value)) return "int";
    if (typeof value === "number") return "float";
    if (typeof value === "string") return "string";
    if (Array.isArray(value) && value.length && value.every((item) => Number.isInteger(item) && item > 0)) return "shape";
    return "json";
  }
